module Base
{
    /* Salvaged from fridges via the Disassemble option on a picked-up fridge.
       See lua/client/CommonTemplates/Moveables/ISStripRefrigeratorAction.lua */
    item RefrigerationComponents
    {
        DisplayCategory = Electronics,
        ItemType = base:normal,
        Weight = 1.0,
        DisplayName = Refrigeration Components,
        Tooltip = Tooltip_item_StrippedFromRefrigerators,
        Icon = ScrapMetal,
        MetalValue = 5.0,
        WorldStaticModel = ElectronicsScrap,
    }

    /* Extra dismantling targets vanilla does not cover. Vanilla's own
       DismantleElectronicsDevice already handles TVs, radios and walkie-talkies,
       so those are deliberately not repeated here. */

    craftRecipe VAEDismantleToyCar
    {
        timedAction = DismantleElectrical,
        time = 80,
        SkillRequired = Electricity:1,
        category = Electrical,
        Tags = InHandCraft;Electrical,
        xpAward = Electricity:10,
        inputs
        {
            item 1 [Base.ToyCar] flags[NoBrokenItems] mode:destroy,
            item 1 tags[base:screwdriver] mode:keep flags[NoBrokenItems],
        }
        outputs
        {
            item 2 Base.ElectronicsScrap,
        }
    }

    craftRecipe VAEDismantleElectronicComponents
    {
        timedAction = DismantleElectrical,
        time = 80,
        SkillRequired = Electricity:3,
        category = Electrical,
        Tags = InHandCraft;Electrical,
        xpAward = Electricity:10,
        inputs
        {
            item 1 [Base.RadioReceiver;Base.RadioTransmitter;Base.ScannerModule;Base.Amplifier] flags[NoBrokenItems] mode:destroy,
            item 1 tags[base:screwdriver] mode:keep flags[NoBrokenItems],
        }
        outputs
        {
            item 2 Base.ElectronicsScrap,
        }
    }

    /* B42 consolidated the coloured guitar variants into GuitarElectric and
       GuitarElectricBass, so the old per-colour item ids no longer resolve. */
    craftRecipe VAEDismantleGuitar
    {
        timedAction = DismantleElectrical,
        time = 120,
        SkillRequired = Electricity:4,
        category = Electrical,
        Tags = InHandCraft;Electrical,
        xpAward = Electricity:25,
        inputs
        {
            item 1 [Base.GuitarElectric;Base.GuitarElectricBass;Base.Keytar] flags[NoBrokenItems] mode:destroy,
            item 1 tags[base:screwdriver] mode:keep flags[NoBrokenItems],
        }
        outputs
        {
            item 4 Base.ElectronicsScrap,
        }
    }
}
